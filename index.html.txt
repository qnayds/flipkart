        const menu = document.getElementById('mobile-menu');
        const icon = document.getElementById('menu-icon');
        isMenuOpen = !isMenuOpen;
        if (isMenuOpen) {
          menu.classList.remove('hidden');
          menu.classList.add('flex', 'animate-fadeIn');
          icon.setAttribute('data-lucide', 'x');
        } else {

          menu.classList.remove('flex', 'animate-fadeIn');
          icon.setAttribute('data-lucide', 'menu');
        }
        lucide.createIcons();
      }

      // 2. PROFILE IMAGE AND ADJUSTMENTS CONTROLLER
      let zoom = 100;
      let offsetX = 0;
      let offsetY = 0;
      let fitMode = 'cover';
      let showControls = false;

      function initProfileImage() {
        // Load settings from localStorage
        const savedPic = localStorage.getItem('sawad_profile_pic');
        const savedZoom = localStorage.getItem('sawad_profile_zoom');
        const savedX = localStorage.getItem('sawad_profile_x');
        const savedY = localStorage.getItem('sawad_profile_y');
        const savedFit = localStorage.getItem('sawad_profile_fit');

        if (savedPic) {
          document.getElementById('profile-img').src = savedPic;
        }
        if (savedZoom) zoom = parseInt(savedZoom);
        if (savedX) offsetX = parseInt(savedX);
        if (savedY) offsetY = parseInt(savedY);
        if (savedFit) fitMode = savedFit;

        // Sync inputs
        document.getElementById('zoom-slider').value = zoom;
        document.getElementById('offset-x-slider').value = offsetX;
        document.getElementById('offset-y-slider').value = offsetY;

        updateImageStyle();
      }

      function handleProfileUpload(event) {
        const file = event.target.files[0];
        if (file) {
          const reader = new FileReader();
          reader.onload = function(e) {
            document.getElementById('profile-img').src = e.target.result;
            localStorage.setItem('sawad_profile_pic', e.target.result);
            
            // Instantly open controls panel for immediate alignment
            showControls = true;
            syncControlsPanel();
          }
          reader.readAsDataURL(file);
        }
      }

      function togglePhotoControls() {
        showControls = !showControls;
        syncControlsPanel();
      }

      function syncControlsPanel() {
        const panel = document.getElementById('photo-controls');
        const btn = document.getElementById('tune-btn');
        const btnText = document.getElementById('tune-btn-text');
        
        if (showControls) {
          panel.classList.remove('hidden');
          btn.classList.add('bg-[#10B981]', 'border-[#10B981]', 'text-white');
          btn.classList.remove('bg-white', 'text-slate-700');
          btnText.classList.remove('hidden');
        } else {
          panel.classList.add('hidden');
          btn.classList.remove('bg-[#10B981]', 'border-[#10B981]', 'text-white');
          btn.classList.add('bg-white', 'text-slate-700');
          btnText.classList.add('hidden');
        }
      }

      function updateImageStyle() {
        zoom = parseInt(document.getElementById('zoom-slider').value);
        offsetX = parseInt(document.getElementById('offset-x-slider').value);
        offsetY = parseInt(document.getElementById('offset-y-slider').value);

        // Update value text tags
        document.getElementById('zoom-val').innerText = zoom + '%';
        document.getElementById('offset-x-val').innerText = offsetX + 'px';
        document.getElementById('offset-y-val').innerText = offsetY + 'px';

        // Apply to style
        const img = document.getElementById('profile-img');
        img.style.objectFit = fitMode;
        img.style.transform = `translate(${offsetX}px, ${offsetY}px) scale(${zoom / 100})`;

        // Persist
        localStorage.setItem('sawad_profile_zoom', zoom);
        localStorage.setItem('sawad_profile_x', offsetX);
        localStorage.setItem('sawad_profile_y', offsetY);
        localStorage.setItem('sawad_profile_fit', fitMode);
      }

      function setFitMode(mode) {
        fitMode = mode;

      function scrollCarousel(direction) {
        const scrollAmount = carousel.clientWidth * 0.75;

      function checkCarouselScroll() {
        const prevBtn = document.getElementById('prev-btn');
        const nextBtn = document.getElementById('next-btn');
        const left = carousel.scrollLeft;
        const max = carousel.scrollWidth - carousel.clientWidth;

        if (left > 5) {
          prevBtn.className = 'h-11 w-11 rounded-full border border-slate-300 bg-white text-slate-700 hover:bg-slate-50 hover:border-slate-400 active:scale-95 shadow-xs flex items-center justify-center transition-all cursor-pointer';
          prevBtn.disabled = false;
        } else {
          prevBtn.className = 'h-11 w-11 rounded-full border border-slate-200 bg-slate-50 text-slate-300 flex items-center justify-center transition-all cursor-not-allowed';
          prevBtn.disabled = true;
        }

        if (left < max - 5) {
          nextBtn.className = 'h-11 w-11 rounded-full border border-slate-300 bg-white text-slate-700 hover:bg-slate-50 hover:border-slate-400 active:scale-95 shadow-xs flex items-center justify-center transition-all cursor-pointer';
          nextBtn.disabled = false;
        } else {
          nextBtn.className = 'h-11 w-11 rounded-full border border-slate-200 bg-slate-50 text-slate-300 flex items-center justify-center transition-all cursor-not-allowed';
          nextBtn.disabled = true;
        }
      }

      function simulateSandboxOutput() {
        if (typewriterInterval) clearInterval(typewriterInterval);
        
        const outputBox = document.getElementById('typewriter-output');
        const simulateBtn = document.getElementById('simulate-btn');
        outputBox.innerText = '';
        outputBox.classList.add('cursor-blink');
        simulateBtn.disabled = true;
        simulateBtn.innerText = 'Generating output...';

        const text = presets[activeCategory].result;
        let index = 0;
        
        // Fast typewriter typing simulator
        typewriterInterval = setInterval(() => {
          if (index < text.length) {
            outputBox.innerText = text.substring(0, index + 15);
            index += 15;
            outputBox.scrollTop = outputBox.scrollHeight;
          } else {
            clearInterval(typewriterInterval);
            outputBox.classList.remove('cursor-blink');
            simulateBtn.disabled = false;
            simulateBtn.innerText = 'Simulate Response';
          }
        }, 12);
      }
